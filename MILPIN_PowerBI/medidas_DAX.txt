== TABLA: Historial_Riego ==

-- KPI 1: Consumo promedio m³/ha del ciclo más reciente
Consumo Agua m3_ha =
CALCULATE(
    AVERAGE(Historial_Riego[volumen_m3_ha]),
    Historial_Riego[ciclo_agricola] = MAX(Historial_Riego[ciclo_agricola])
)

-- Línea base y objetivo (constantes DR-041)
Linea Base m3_ha = 8000
Objetivo m3_ha   = 6000
Tarifa MXN_m3    = 1.68

-- KPI 2: Ahorro en m³/ha
Ahorro m3_ha =
[Linea Base m3_ha] - [Consumo Agua m3_ha]

-- KPI 3: Ahorro en costo MXN/ha
Ahorro Costo MXN_ha =
[Ahorro m3_ha] * [Tarifa MXN_m3]

-- % de ahorro vs. línea base
Pct Ahorro =
DIVIDE([Ahorro m3_ha], [Linea Base m3_ha], 0)

-- Total de agua aplicada en el período filtrado
Total Agua Aplicada m3 =
SUMX(
    Historial_Riego,
    Historial_Riego[volumen_m3_ha] * RELATED(Parcelas[area_ha])
)

-- Costo total de energía
Costo Total Energia MXN =
SUM(Historial_Riego[costo_energia_mxn])

-- Eventos sobre la meta (> 6,500 m³/ha)
Eventos Sobre Meta =
COUNTROWS(
    FILTER(Historial_Riego, Historial_Riego[volumen_m3_ha] > 6500)
)

-- Consumo promedio por método de riego
Consumo Promedio por Metodo =
AVERAGE(Historial_Riego[volumen_m3_ha])


== TABLA: Recomendaciones ==

-- KPI 4: Tasa de adopción (% aceptadas)
Tasa Adopcion Pct =
DIVIDE(
    COUNTROWS(FILTER(Recomendaciones, Recomendaciones[aceptada] = "aceptada")),
    COUNTROWS(Recomendaciones),
    0
)

-- Conteos por estado de recomendación
Total Recomendaciones    = COUNTROWS(Recomendaciones)
Rec Aceptadas            = COUNTROWS(FILTER(Recomendaciones, Recomendaciones[aceptada] = "aceptada"))
Rec Pendientes           = COUNTROWS(FILTER(Recomendaciones, Recomendaciones[aceptada] = "pendiente"))
Rec Ignoradas            = COUNTROWS(FILTER(Recomendaciones, Recomendaciones[aceptada] = "ignorada"))

-- ETc promedio calculada (para gráfico de barras por cultivo)
ETc Promedio mm =
AVERAGE(Recomendaciones[etc_calculada])

-- ETo promedio de referencia
ETo Promedio =
AVERAGE(Recomendaciones[eto_referencia])

-- Déficit hídrico acumulado promedio
Deficit Acumulado Promedio =
AVERAGE(Recomendaciones[deficit_acumulado_mm])

-- Urgencia crítica: % de recomendaciones en nivel crítico
Pct Urgencia Critica =
DIVIDE(
    COUNTROWS(FILTER(Recomendaciones, Recomendaciones[nivel_urgencia] = "critico")),
    COUNTROWS(Recomendaciones),
    0
)


== TABLA: Parcelas ==

-- Total hectáreas activas
Total Hectareas =
CALCULATE(
    SUM(Parcelas[area_ha]),
    Parcelas[activo] = TRUE()
)

-- Parcelas activas
Total Parcelas Activas =
CALCULATE(
    COUNTROWS(Parcelas),
    Parcelas[activo] = TRUE()
)

-- % superficie por sistema de riego (usar en gráfico donut)
Pct Area por Sistema =
DIVIDE(
    SUM(Parcelas[area_ha]),
    CALCULATE(SUM(Parcelas[area_ha]), ALL(Parcelas[sistema_riego])),
    0
)


== TABLA: Costos_Ciclo ==

-- Ingreso estimado total
Ingreso Total MXN =
SUM(Costos_Ciclo[ingreso_estimado_mxn])

-- Margen de contribución promedio por ciclo
Margen Contribucion Promedio =
AVERAGE(Costos_Ciclo[margen_contribucion_mxn])

-- Costo total de agua por ciclo
Costo Total Agua MXN =
SUM(Costos_Ciclo[costo_agua_mxn])


======================================================
  FORMATO SUGERIDO PARA TARJETAS KPI
======================================================

KPI 1 — Consumo de Agua
  Valor   : [Consumo Agua m3_ha]   → Formato: #,##0 "m³/ha"
  Tendencia: [Pct Ahorro]          → Formato: -0% (color verde si positivo)

KPI 2 — Ahorro Estimado
  Valor   : [Ahorro m3_ha]         → Formato: #,##0 "m³/ha"
  Tendencia: [Pct Ahorro]          → Formato: +0% eficiencia

KPI 3 — Ahorro en Costo
  Valor   : [Ahorro Costo MXN_ha]  → Formato: $#,##0 "MXN/ha"
  Subtítulo: "Tarifa CFE 9-CU · $1.68 MXN/m³"

KPI 4 — Tasa de Adopción
  Valor   : [Tasa Adopcion Pct]    → Formato: 0% "aceptadas"
  Comparar vs ciclo anterior manualmente o con variable SELECTEDVALUE
